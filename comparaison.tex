\section{Quand utiliser Git ?}
\begin{frame}
    \frametitle{Quand utiliser Git ?}
    \begin{itemize}
        \item Gestion de code source ou fichiers texte plus généralement
        \item Travail en équipe technique avec suivi précis des modifications
        \item Historique complet et décentralisé
        \item \textbf{Non adapté} aux fichiers de plus de 2Go sans outils supplémentaires (Git Large File System)
    \end{itemize}
\end{frame}

\section{Recommandations}
\begin{frame}
    \frametitle{Recommandations}
    \begin{block}{Grand public - Bureautique}
    \begin{itemize}
    	\item OneDrive si tu utilises Office
    	\item Google Drive si tu utilises Google Workspace
    \end{itemize}
    \end{block}
    \begin{block}{Professionnels - Équipe, projet administratif}
    \begin{itemize}
    	\item Dropbox pour les intégrations
    	\item OneDrive si tu es dans l’écosystème Microsoft
    \end{itemize}
    \end{block}
    \begin{block}{Professionnels - Équipe, projet technique ou mixte}
	 \begin{itemize}
    	\item Git (e.g. GitHub, GitLab, Bitbucket)
    \end{itemize}    
    \end{block}
\end{frame}
